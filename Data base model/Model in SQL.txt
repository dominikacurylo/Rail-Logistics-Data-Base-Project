CREATE TABLE Diesel_locomotive (

  Frequency      number(10) NOT NULL, 
  PRIMARY KEY (ElectricTypeID));
CREATE TABLE Facility (
  ID          number(38) GENERATED AS IDENTITY, 
  Name        varchar2(255) NOT NULL UNIQUE, 
  TypeID      number(10) NOT NULL, 
  Address     varchar2(255) UNIQUE, 
  City        varchar2(255), 
  Capacity    number(10), 
  Electrified char(3), 
  OwnerVAT    varchar2(15) NOT NULL, 
  PRIMARY KEY (ID), 
  CONSTRAINT CK_Facility_Electrified 
    CHECK (Electrified IN ('Yes','No')));
CREATE TABLE FacilityType (
  FacilityTypeID number(10) GENERATED AS IDENTITY, 
  Name           varchar2(255) NOT NULL, 
  PRIMARY KEY (FacilityTypeID));
CREATE TABLE Gauge (
  Name  varchar2(255) NOT NULL, 
  Gauge number(10) NOT NULL, 
  PRIMARY KEY (Name));
CREATE TABLE Line (
  ID        number(38) GENERATED AS IDENTITY, 
  Name      varchar2(255) NOT NULL UNIQUE, 
  OwnerVAT  varchar2(15) NOT NULL, 
  StartID   number(38) NOT NULL, 
  EndID     number(38) NOT NULL, 
  GaugeName varchar2(255) NOT NULL, 
  PRIMARY KEY (ID));
CREATE TABLE Locomotive (
  LocomotiveNumber number(38) NOT NULL, 
  Name             varchar2(255) NOT NULL, 
  ModelID          number(38) NOT NULL, 
  Service          number(4) NOT NULL, 
  OperatorVAT      varchar2(15) NOT NULL, 
  PRIMARY KEY (LocomotiveNumber));
CREATE TABLE LocomotiveModel (
  ID               number(38) GENERATED AS IDENTITY, 
  Make             varchar2(255) NOT NULL, 
  Model            varchar2(255) NOT NULL, 
  NumberBogies     number(10) NOT NULL, 
  Bogies           varchar2(255) NOT NULL, 
  Power            number(10) NOT NULL, 
  Length           float(10) NOT NULL, 
  Width            float(10) NOT NULL, 
  Height           number(10) NOT NULL, 
  Weight           number(10) NOT NULL, 
  MaxSpeed         number(10) NOT NULL, 
  OperationalSpeed float(10) NOT NULL, 
  Traction         number(10) NOT NULL, 
  DieselTypeID     number(38), 
  ElectricTypeID   number(38), 
  GaugeName        varchar2(255) NOT NULL, 
  PRIMARY KEY (ID), 
  CONSTRAINT CK_LocomotiveModel_OneType 
    CHECK ((DieselTypeID IS NOT NULL) != (ElectricTypeID IS NOT NULL) ));
CREATE TABLE Operator (

  PRIMARY KEY (VATNumber));
CREATE TABLE Owner (

  Name      varchar2(255) NOT NULL,

  PRIMARY KEY (VATnumber));
CREATE TABLE Segment (
  ID             number(38) NOT NULL, 
  LineID         number(38) NOT NULL,

  Electrified    char(3) NOT NULL, 
  Max_weight     float(38) NOT NULL, 
  Length         number(38) NOT NULL, 
  NumberOfTracks number(10) NOT NULL, 
  SpeedLimit     number(10), 
  PRIMARY KEY (ID), 
  CONSTRAINT CK_LineSegment_Electrified 
    CHECK (Electrified IN ('Yes','No')));
