CREATE OR REPLACE FUNCTION fn_register_model_gauge(
    p_ModelID   IN NUMBER,
    p_GaugeName IN VARCHAR2,
    p_GaugeSize IN NUMBER
) RETURN VARCHAR2 IS
BEGIN

    IF p_GaugeSize <= 0 THEN
        RETURN 'ERROR_NEGATIVE_GAUGE';
    END IF;

    BEGIN
        INSERT INTO Gauge (Name, Gauge) VALUES (p_GaugeName, p_GaugeSize);
    EXCEPTION WHEN DUP_VAL_ON_INDEX THEN NULL; END;

    INSERT INTO Gauge_WagonModel (GaugeName, WagonModelModelNumber) 
    VALUES (p_GaugeName, p_ModelID);
    
    RETURN 'GAUGE_ADDED';
EXCEPTION
    WHEN DUP_VAL_ON_INDEX THEN RETURN 'GAUGE_ADDED';
    WHEN OTHERS THEN RETURN 'ERROR';
END;
/

CREATE OR REPLACE FUNCTION fn_register_wagon(
    p_WagonNumber IN VARCHAR2,
    p_ModelID     IN NUMBER,
    p_Year        IN NUMBER,
    p_Operator    IN VARCHAR2
) RETURN VARCHAR2 IS
    v_check NUMBER;
BEGIN
    SELECT COUNT(*) INTO v_check FROM Gauge_WagonModel WHERE WagonModelModelNumber = p_ModelID;
    
    IF v_check = 0 THEN RETURN 'ERROR_NO_GAUGE'; END IF;

    INSERT INTO Wagon (WagonNumber, ModelID, ServiceYear, OperatorShortName)
    VALUES (p_WagonNumber, p_ModelID, p_Year, p_Operator);
    
    RETURN 'WAGON_ADDED';
EXCEPTION
    WHEN DUP_VAL_ON_INDEX THEN RETURN 'ERROR_WAGON_ALREADY_EXISTS';
    WHEN OTHERS THEN RETURN 'ERROR';
END;
/
