CREATE OR REPLACE TRIGGER trg_check_train_max_length
BEFORE INSERT OR UPDATE ON Wagon_Freight
FOR EACH ROW
DECLARE
    v_current_total_length NUMBER := 0;
    v_max_allowed_length   NUMBER;
    v_new_wagon_length     NUMBER;
    v_train_id             NUMBER;
BEGIN
    -- identify the train linked to this specific freight
    SELECT TrainID INTO v_train_id 
    FROM Train_Freight 
    WHERE FreightID = :NEW.FreightID;

    -- get the maximum length defined for this train
    SELECT MaxLength INTO v_max_allowed_length 
    FROM Train 
    WHERE ID = v_train_id;

    -- calculate current total length
    SELECT NVL(SUM(wm.Length), 0) INTO v_current_total_length
    FROM Wagon_Freight wf
    JOIN Wagon w ON wf.WagonWagonNumber = w.WagonNumber
    JOIN WagonModel wm ON w.ModelID = wm.ModelNumber
    WHERE wf.FreightID IN (SELECT FreightID FROM Train_Freight WHERE TrainID = v_train_id);

    -- get the length of the new wagon being added
    SELECT wm.Length INTO v_new_wagon_length
    FROM Wagon w
    JOIN WagonModel wm ON w.ModelID = wm.ModelNumber
    WHERE w.WagonNumber = :NEW.WagonWagonNumber;

-- checking
    IF (v_current_total_length + v_new_wagon_length) > v_max_allowed_length THEN
        RAISE_APPLICATION_ERROR(-20010, 
            'LIMIT EXCEEDED: Train MaxLength is ' || v_max_allowed_length || 
            'm. Current total with this wagon would be ' || (v_current_total_length + v_new_wagon_length) || 'm.');
    END IF;

EXCEPTION
    WHEN NO_DATA_FOUND THEN
        NULL;
END;
