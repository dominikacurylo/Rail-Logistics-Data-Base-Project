CREATE OR REPLACE FUNCTION fn_register_actual_route(
    p_TrainID       IN NUMBER,
    p_LineID        IN NUMBER,
    p_Direction     IN VARCHAR2,
    p_DepartureTime IN DATE,
    p_ArrivalTime   IN DATE DEFAULT NULL
) RETURN VARCHAR2 IS
BEGIN

    INSERT INTO ActualRoute (
        TrainID, 
        LineID, 
        Direction, 
        DepartureTime, 
        ArrivalTime
    ) VALUES (
        p_TrainID, 
        p_LineID, 
        p_Direction, 
        p_DepartureTime, 
        p_ArrivalTime
    );
    
    RETURN 'SUCCESS: Movement registered for Train ' || p_TrainID || ' on Line ' || p_LineID;

EXCEPTION
    WHEN DUP_VAL_ON_INDEX THEN
        RETURN 'ERROR: Movement already exists for this Train and Line.';
    WHEN OTHERS THEN
       IF p_ArrivalTime IS NOT NULL AND p_ArrivalTime < p_DepartureTime THEN
             RETURN 'ERROR: Arrival time cannot be earlier than Departure time.';
        ELSE
             RETURN 'ERROR: ' || SQLERRM;
        END IF;
END;
