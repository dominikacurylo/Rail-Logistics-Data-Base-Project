Facility.TypeID%TYPE,    p_owner_short    IN Facility.OwnerShortName%TYPE,    p_address        IN Facility.Address%TYPE,    p_city           IN Facility.City%TYPE,    p_capacity       IN Facility.Capacity%TYPE,    p_electrified    IN Facility.Electrified%TYPE) RETURN SYS_REFCURSOR IS    v_cursor SYS_REFCURSOR;    v_new_id Facility.ID%TYPE;BEGIN    SELECT NVL(MAX(ID), 0) + 1 INTO v_new_id FROM Facility;    INSERT INTO Facility (ID, Name, TypeID, OwnerShortName, Address, City, Capacity, Electrified)    VALUES (v_new_id, p_name, p_type_id, p_owner_short, p_address, p_city, p_capacity, p_electrified);    OPEN v_cursor FOR        SELECT * FROM Facility WHERE ID = v_new_id;    RETURN v_cursor;EXCEPTION    WHEN OTHERS THEN        RAISE_APPLICATION_ERROR(-20001, 'Error
