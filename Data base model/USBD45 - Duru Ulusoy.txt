A line must have at least one segment.CREATE OR REPLACE FUNCTION fnc_add_line_with_segment (    p_line_id         IN Line.ID%TYPE,    p_line_name       IN Line.Name%TYPE,    p_start_id        IN Line.StartID%TYPE,    p_end_id          IN Line.EndID%TYPE,    p_owner_short     IN Line.OwnerShortName%TYPE,    p_gauge_name      IN Line.GaugeName%TYPE,    p_seg_order       IN Segment."Order"%TYPE,     p_seg_electrified IN Segment.Electrified%TYPE,    p_seg_max_weight  IN Segment.Max_weight%TYPE,    p_seg_length      IN Segment.Length%TYPE,    p_seg_tracks      IN Segment.NumberOfTracks%TYPE,    p_seg_speed       IN Segment.SpeedLimit%TYPE) RETURN SYS_REFCURSOR IS    v_cursor SYS_REFCURSOR;    v_seg_id Segment.ID%TYPE;BEGIN    INSERT INTO Line (ID, Name, StartID, EndID, OwnerShortName, GaugeName)    VALUES (p_line_id, p_line_name, p_start_id, p_end_id, p_owner_short, p_gauge_name);    SELECT NVL(MAX(ID), 0) + 1 INTO v_seg_id FROM Segment;    INSERT INTO Segment (ID, "Order", Electrified, Max_weight, Length, NumberOfTracks, SpeedLimit)    VALUES (v_seg_id, p_seg_order, p_seg_electrified, p_seg_max_weight, p_seg_length, p_seg_tracks, p_seg_speed);    INSERT INTO Line_Segment (LineID, SegmentID)    VALUES (p_line_id, v_seg_id);    OPEN v_cursor FOR        SELECT l.ID as LineID, l.Name as LineName, s.ID as SegmentID, s."Order", s.Length        FROM Line l        JOIN Line_Segment ls ON l.ID = ls.LineID        JOIN Segment s ON ls.SegmentID = s.ID        WHERE l.ID = p_line_id;        RETURN v_cursor;EXCEPTION    WHEN OTHERS THEN        ROLLBACK;        RAISE_APPLICATION_ERROR(-20002, 'Error adding line with segment: ' || SQLERRM);END;/---------------------------------------TESTS---------------------------------------SET SERVEROUTPUT ON;SET FEEDBACK OFF;DECLARE    v_cursor     SYS_REFCURSOR;
