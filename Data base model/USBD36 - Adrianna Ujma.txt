CREATE OR REPLACE FUNCTION AddBuildingToFacility(
    p_FacilityID IN NUMBER,
    p_BuildingTypeID IN NUMBER,
    p_Count IN NUMBER
) RETURN VARCHAR2 IS
    v_exists NUMBER;
BEGIN
      SELECT COUNT(*) INTO v_exists FROM Facility WHERE ID = p_FacilityID;
    IF v_exists = 0 THEN
        RETURN 'FACILITY_NOT_FOUND';
    END IF;

    SELECT COUNT(*) INTO v_exists FROM BuildingType WHERE ID = p_BuildingTypeID;
    IF v_exists = 0 THEN
        RETURN 'BUILDING_TYPE_NOT_FOUND';
    END IF;

    UPDATE Facility_building
    SET Count = Count + p_Count
    WHERE FacilityID = p_FacilityID AND BuildingTypeID = p_BuildingTypeID;

 
    IF SQL%NOTFOUND THEN
        INSERT INTO Facility_building (FacilityID, BuildingTypeID, Count)
        VALUES (p_FacilityID, p_BuildingTypeID, p_Count);
    END IF;

    COMMIT;
    RETURN 'OK';
EXCEPTION
    WHEN OTHERS THEN
        ROLLBACK;
        RETURN 'ERROR';
END;
/
