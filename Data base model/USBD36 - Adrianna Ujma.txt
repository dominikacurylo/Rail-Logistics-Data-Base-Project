CREATE OR REPLACE FUNCTION AddBuildingToFacility(
    p_FacilityID IN NUMBER,
    p_BuildingTypeID IN NUMBER,
    p_Count IN NUMBER
) RETURN VARCHAR2 IS
    v_exists NUMBER;
BEGIN
    IF p_Count <= 0 THEN
        RETURN 'INVALID';
    END IF;

    SELECT COUNT(*) INTO v_exists FROM Facility WHERE ID = p_FacilityID;
    IF v_exists = 0 THEN
        RETURN 'FACILITY_NOT_FOUND';
    END IF;

    SELECT COUNT(*) INTO v_exists FROM BuildingType WHERE ID = p_BuildingTypeID;
    IF v_exists = 0 THEN
        RETURN 'BUILDING_TYPE_NOT_FOUND';
    END IF;

    UPDATE Facility_building
    SET Count = Count + p_Count
    WHERE FacilityID = p_FacilityID AND BuildingTypeID = p_BuildingTypeID;

    IF SQL%NOTFOUND THEN
        INSERT INTO Facility_building (FacilityID, BuildingTypeID, Count)
        VALUES (p_FacilityID, p_BuildingTypeID, p_Count);
    END IF;

    COMMIT;
    RETURN 'OK';

EXCEPTION
    WHEN OTHERS THEN
        ROLLBACK;
        RETURN 'ERROR';
END AddBuildingToFacility;

DECLARE
    v_result      VARCHAR2(100);
    v_FacID       NUMBER := 100;
    v_BldTypeID   NUMBER := 10;
    
    PROCEDURE Report_Test(p_Scenario VARCHAR2, p_Expected VARCHAR2, p_Actual VARCHAR2) IS
    BEGIN
        IF p_Expected = p_Actual THEN
            DBMS_OUTPUT.PUT_LINE('[PASS] ' || p_Scenario || '; Result: ' || p_Actual);
        ELSE
            DBMS_OUTPUT.PUT_LINE('[FAIL] ' || p_Scenario || '; Expected: ' || p_Expected || ', Received: ' || p_Actual);
        END IF;
    END Report_Test;

BEGIN
    DBMS_OUTPUT.PUT_LINE('======= USBD36 data setup =======');


    BEGIN
        INSERT INTO Owner (ShortName, Name, VATnumber) 
        VALUES ('IP', 'Infraestruturas de Portugal, SA', 'PT503933813');
    EXCEPTION WHEN DUP_VAL_ON_INDEX THEN NULL; END;

    BEGIN
        INSERT INTO FacilityType (FacilityTypeID, Name) VALUES (5, 'Railway Station');
    EXCEPTION WHEN DUP_VAL_ON_INDEX THEN NULL; END;

    BEGIN
        INSERT INTO Facility (ID, Name, TypeID, Address, City, Capacity, Electrified, OwnerShortName)
        VALUES (v_FacID, 'Barroselas', 5, 'Rua da Estacao', 'Barroselas', 50, 'Yes', 'IP');
    EXCEPTION WHEN DUP_VAL_ON_INDEX THEN NULL; END;


    BEGIN
        INSERT INTO BuildingType (ID, Type) VALUES (v_BldTypeID, 'Warehouse');
    EXCEPTION WHEN DUP_VAL_ON_INDEX THEN NULL; END;

    COMMIT;

    DBMS_OUTPUT.PUT_LINE('====== USBD36 tests ======');

    -- Case 1: adding buildings correctly
    v_result := AddBuildingToFacility(v_FacID, v_BldTypeID, 5);
    Report_Test('Add Building Valid', 'OK', v_result);

    -- Case 2: negative count 
    v_result := AddBuildingToFacility(v_FacID, v_BldTypeID, -10);
    Report_Test('Negative Count Check', 'INVALID', v_result);

    -- Case 3: non-existing facility
    v_result := AddBuildingToFacility(9999, v_BldTypeID, 1);
    Report_Test('Invalid Facility ID', 'FACILITY_NOT_FOUND', v_result);

    -- Case 4: non-existing building type
    v_result := AddBuildingToFacility(v_FacID, 9999, 1);
    Report_Test('Invalid Building Type', 'BUILDING_TYPE_NOT_FOUND', v_result);

    DBMS_OUTPUT.PUT_LINE('======= End of tests =======');
END;
